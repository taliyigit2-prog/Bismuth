import { appendFileSync } from 'node:fs'
const log = (...a: unknown[]) => appendFileSync('/tmp/fable/repro/log.txt', a.join(' ') + '\n')
import { afterEach, describe, expect, it } from 'vitest'
import { eq } from 'drizzle-orm'
import { depsOf, roleCtx, testClock } from '../../../tests/factories/ctx'
import { seededDatabase } from '../../../tests/factories/db'
import type { DatabaseHandle } from '../db/client'
import { paymentMethods, roomTypes, salesChannels } from '../db/schema'
import { createServices } from './index'

const owner = roleCtx('owner')
const reception = roleCtx('reception')
let handle: DatabaseHandle
afterEach(() => handle?.close())

async function setup(now = '2026-10-01T08:00:00.000Z') {
  handle = await seededDatabase()
  const clock = testClock(now)
  const deps = depsOf(handle, clock)
  const svc = createServices(
    handle,
    { name: 'B', version: 't', packaged: false, dataDir: '/tmp/unused', electron: null, chrome: null },
    { clock: deps.clock }
  )
  const std = handle.db.select().from(roomTypes).where(eq(roomTypes.code, 'standard')).get()!
  const r501 = svc.rooms.createRoom(owner, { number: '501', roomTypeId: std.id, beds: 2, validFrom: '2026-01-01' })
  const direct = handle.db.select().from(salesChannels).where(eq(salesChannels.code, 'direct')).get()!.id
  const cash = handle.db.select().from(paymentMethods).where(eq(paymentMethods.code, 'cash')).get()!.id
  const book = (ci: string, co: string, name = 'Örnek') =>
    svc.reservations.create(reception, {
      guest: { firstName: name, lastName: 'Misafir', phone: '0532 000 00 00' },
      channelId: direct,
      stays: [{ roomId: r501.id, checkIn: ci, checkOut: co, nightlyPriceMinor: 200_000 }]
    })
  return { svc, clock, r501, cash, book }
}
const grid = (svc: any) => svc.calendar.grid(owner, { from: '2026-10-01', to: '2026-10-10' })

describe('1) check-in sonrası iptal / iade', () => {
  it('in_house iken iptal reddedilir (guestInHouse)', async () => {
    const { svc, book } = await setup()
    const r = book('2026-10-01', '2026-10-03')
    svc.reservations.checkIn(reception, r.stays[0]!.id)
    expect(() => svc.reservations.cancel(reception, r.reservation.id, 'beğenmedi')).toThrow(/guestInHouse/)
  })
  it('checked_out sonrası iptal reddedilir (reservationClosed); iade ancak payments.refund ile', async () => {
    const { svc, clock, book, cash } = await setup()
    const r = book('2026-10-01', '2026-10-03')
    svc.payments.create(reception, { reservationId: r.reservation.id, date: '2026-10-01', kind: 'payment', methodId: cash, amountMinor: 200_000, currency: 'TRY' })
    svc.reservations.checkIn(reception, r.stays[0]!.id)
    clock.set('2026-10-01T10:00:00.000Z')
    svc.reservations.checkOut(reception, r.stays[0]!.id)
    expect(() => svc.reservations.cancel(reception, r.reservation.id, 'beğenmedi')).toThrow(/reservationClosed/)
    expect(() => svc.payments.create(reception, { reservationId: r.reservation.id, date: '2026-10-01', kind: 'refund', methodId: cash, amountMinor: 100_000, currency: 'TRY' })).toThrow(/PERMISSION_DENIED/)
    const mgr = roleCtx('manager')
    svc.payments.create(mgr, { reservationId: r.reservation.id, date: '2026-10-01', kind: 'refund', methodId: cash, amountMinor: 100_000, currency: 'TRY' })
  })
})

describe('3) ödemeli rezervasyon iptali', () => {
  it('iptal: gece+masraf boşalır, ödeme kalır (void değil), bakiye negatif; ızgarada yok', async () => {
    const { svc, book, cash } = await setup()
    const r = book('2026-10-03', '2026-10-05')
    svc.payments.create(reception, { reservationId: r.reservation.id, date: '2026-10-01', kind: 'deposit', methodId: cash, amountMinor: 200_000, currency: 'TRY' })
    const out = svc.reservations.cancel(reception, r.reservation.id, 'vazgeçti')
    log('CANCEL', out.reservation.status, JSON.stringify(out.folio), out.stays.map((s) => s.status), out.stays.map((s) => s.nights.length))
    expect(grid(svc).stays.find((s: any) => s.reservationId === r.reservation.id)).toBeUndefined()
    log('GRID stays', grid(svc).stays.length)
  })
})

describe('4) aynı gün oda tekrar satışı', () => {
  it('A: 1 gecelik konaklama sabah giriş, akşam çıkış -> aynı gece yeniden satılır', async () => {
    const { svc, clock, book } = await setup()
    const r = book('2026-10-01', '2026-10-02')
    svc.reservations.checkIn(reception, r.stays[0]!.id)
    // çıkıştan ÖNCE satış
    expect(() => book('2026-10-01', '2026-10-02', 'Ikinci')).toThrow(/roomOccupied/)
    clock.set('2026-10-01T15:00:00.000Z')
    svc.reservations.checkOut(reception, r.stays[0]!.id)
    expect(book('2026-10-01', '2026-10-02', 'Ikinci').stays[0]!.nights).toHaveLength(1)
  })
  it('B: ikinci gündüz kullanımı (aynı oda aynı gün) reddedilir', async () => {
    const { book } = await setup()
    book('2026-10-01', '2026-10-01')
    expect(() => book('2026-10-01', '2026-10-01', 'Ikinci')).toThrow(/roomOccupied/)
  })
  it('C: gün içinde ikinci kez giriş-çıkış (gündüz yuvası dolu) -> gece yerinde kalır, oda o gece satılamaz', async () => {
    const { svc, clock, book } = await setup()
    const a = book('2026-10-01', '2026-10-02')
    svc.reservations.checkIn(reception, a.stays[0]!.id)
    clock.set('2026-10-01T12:00:00.000Z')
    svc.reservations.checkOut(reception, a.stays[0]!.id) // gece -> gündüz yuvası
    const b = book('2026-10-01', '2026-10-02', 'Ikinci') // bu gece satıldı
    svc.reservations.checkIn(reception, b.stays[0]!.id)
    clock.set('2026-10-01T17:00:00.000Z')
    const out = svc.reservations.checkOut(reception, b.stays[0]!.id)
    log('C stay', out.stays[0]!.checkIn, out.stays[0]!.checkOut, out.stays[0]!.nights.map((n) => n.slot))
    expect(() => book('2026-10-01', '2026-10-02', 'Ucuncu')).toThrow(/roomOccupied/)
  })
  it('D: gündüz kullanımı + aynı gece satışı birlikte olabilir', async () => {
    const { book } = await setup()
    book('2026-10-01', '2026-10-01')
    expect(book('2026-10-01', '2026-10-02', 'Gece').stays[0]!.nights[0]!.slot).toBe('night')
  })
})
